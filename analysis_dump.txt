        fetchWithAuth('/api/wishflow/history'),
        fetchWithAuth('/api/whatsapp/templates'),
      ]);

      const safeJson = async <T,>(res: Response, fallback: T): Promise<T> => {
        if (!res.ok) return fallback;
        const ct = res.headers.get('content-type');
        if (!ct || !ct.includes('application/json')) return fallback;
        try {
          return await res.json();
        } catch {
          return fallback;
        }
      };

      const sumData = await safeJson(sumRes, null);
      if (sumData) setSummaryList(sumData);

      const setData = await safeJson(setRes, null);
      if (setData) setSettings(setData);

      const histData = await safeJson(histRes, null);
      if (histData) setHistoryList(histData);

      const tmplData = await safeJson(tmplRes, null);
      if (tmplData) setTemplates(tmplData);

      if (retries > 0) {
        setTimeout(() => loadData(retries - 1), 800);

        console.warn('Notice loading Birthday WishFlow data:', err?.message || err);

      setLoading(false);

import React, { useState, useEffect } from 'react';
import { Contact, MessageTemplate } from '../types.ts';

  onAutomationCreated: () => void;
  onNavigate: (tab: string) => void;
}

  onAutomationCreated,
  onNavigate,

  const [contacts, setContacts] = useState<Contact[]>([]);
  const [templates, setTemplates] = useState<MessageTemplate[]>([]);
  const [loading, setLoading] = useState(true);

  // Form states
  const [selectedContactId, setSelectedContactId] = useState<string>('');
  const [selectedTemplateId, setSelectedTemplateId] = useState<string>('');
  const [variables, setVariables] = useState<Record<string, string>>({});
  const [scheduleDate, setScheduleDate] = useState('');
  const [scheduleTime, setScheduleTime] = useState('');
  const [timezone, setTimezone] = useState('Asia/Kolkata');

  // Execution states
  const [submitting, setSubmitting] = useState(false);
  const [errorMessage, setErrorMessage] = useState<string | null>(null);
  const [successData, setSuccessData] = useState<{
    customer: string;
    contactPhone: string;
    scheduled: string;
    template: string;
    status: string;
  } | null>(null);

  // Initialize form with defaults
  useEffect(() => {
    const fetchData = async () => {
      setLoading(true);
      try {
        const [cRes, tRes] = await Promise.all([
          fetchWithAuth('/api/contacts'),
          fetchWithAuth('/api/whatsapp/templates'),
        ]);

        if (cRes.ok) {
          const cData = await cRes.json();
          setContacts(cData);
          if (cData.length > 0) setSelectedContactId(String(cData[0].id));
        }

        if (tRes.ok) {
          const tData = await tRes.json();
          // Filter strictly approved templates
          const approved = tData.filter((t: MessageTemplate) => t.status === 'APPROVED');
          setTemplates(approved);
          if (approved.length > 0) {
            setSelectedTemplateId(String(approved[0].id));
            initVariables(approved[0]);
          }
        }
      } catch (err) {
        console.error('Failed to load contacts/templates:', err);
      } finally {
        setLoading(false);
      }
    };

    fetchData();

    // Default to today + 1 hour in local time
    const now = new Date();
    now.setHours(now.getHours() + 1);
    now.setMinutes(0);
    const dateStr = now.toISOString().split('T')[0];
    const timeStr = now.toTimeString().slice(0, 5); // "HH:MM"
    setScheduleDate(dateStr);
    setScheduleTime(timeStr);

    const count = template.variableCount || 0;
    const initialVars: Record<string, string> = {};
    for (let i = 1; i <= count; i++) {
      if (i === 1) initialVars['1'] = 'Rahul Sharma';
      else if (i === 2) initialVars['2'] = 'Appointment #123';
      else if (i === 3) initialVars['3'] = '15 September 2026';
      else initialVars[String(i)] = '';

    setVariables(initialVars);

    setSelectedTemplateId(templateIdStr);
    const found = templates.find((t) => String(t.id) === templateIdStr);
    if (found) {
      initVariables(found);
    }
  };

  const handleVariableChange = (varKey: string, val: string) => {
    setVariables((prev) => ({
      ...prev,
      [varKey]: val,
    }));
  };

    e.preventDefault();
    setErrorMessage(null);
    setSuccessData(null);
    setSubmitting(true);

      if (!selectedContactId) {
        throw new Error('Please select a customer contact.');
      }
      if (!selectedTemplateId) {
        throw new Error('Please select an approved WhatsApp template.');
      }
      if (!scheduleDate || !scheduleTime) {
        throw new Error('Please select both a date and time for scheduled execution.');
      }

      // Compose ISO string
      const scheduledDateTime = new Date(`${scheduleDate}T${scheduleTime}:00`);
      if (isNaN(scheduledDateTime.getTime())) {
        throw new Error('Invalid scheduled date or time.');
      }

      const res = await fetchWithAuth('/api/automations', {
        method: 'POST',
        body: JSON.stringify({
          contactId: parseInt(selectedContactId, 10),
          templateId: parseInt(selectedTemplateId, 10),
          variables,
          scheduledAt: scheduledDateTime.toISOString(),
          timezone,
        }),
      });

      const data = await res.json();
      if (!res.ok) {
        throw new Error(data.error || 'Failed to schedule automation');
      }

      setSuccessData({
        customer: data.automation.customer,
        contactPhone: data.automation.contactPhone,
        scheduled: new Date(data.automation.scheduled).toLocaleString(undefined, {
          dateStyle: 'medium',

/**
 * Reads and parses all .env files dynamically from disk.
 * This guarantees that when a user creates or edits .env at runtime,
 * the new credentials are immediately detected without restarting Node.
 */
function readEnvFilesFromDisk(): Record<string, string> {
  const envVars: Record<string, string> = {};
  const rootDir = process.cwd();
  const fileCandidates = ['.env', '.env.local', '.env.development', '.env.production'];

  for (const file of fileCandidates) {
    const fullPath = path.join(rootDir, file);
    if (fs.existsSync(fullPath)) {
      try {
        const content = fs.readFileSync(fullPath, 'utf-8');
        const parsed = dotenv.parse(content);
        Object.assign(envVars, parsed);
      } catch (err: any) {
        console.warn(`[Credentials] Notice reading ${file}:`, err?.message);
      }
    }
  }

  return envVars;
}

/**
 * Extracts a value checking both process.env and disk .env files against multiple key aliases.
 */
function getFirstMatch(diskEnv: Record<string, string>, aliases: string[]): { value: string; fromDisk: boolean } {
  // 1. Check process.env first
  for (const alias of aliases) {
    const val = process.env[alias];
    if (val !== undefined && val !== null && String(val).trim() !== '') {
      return { value: String(val).trim().replace(/['"]/g, ''), fromDisk: false };
    }
  }

  // 2. Check disk .env files
  for (const alias of aliases) {
    const val = diskEnv[alias];
    if (val !== undefined && val !== null && String(val).trim() !== '') {
      return { value: String(val).trim().replace(/['"]/g, ''), fromDisk: true };
    }
  }

  return { value: '', fromDisk: false };
}

/**
 * Centrally resolves WhatsApp credentials by combining environment variables (.env files),
 * database records, and local in-memory store.
 */
export async function resolveWhatsAppCredentials(): Promise<ResolvedWhatsAppCredentials> {
  const diskEnv = readEnvFilesFromDisk();

  // 1. Env detection
  const envTokenMatch = getFirstMatch(diskEnv, TOKEN_ALIASES);
  const envPhoneIdMatch = getFirstMatch(diskEnv, PHONE_ID_ALIASES);
  const envWabaIdMatch = getFirstMatch(diskEnv, WABA_ID_ALIASES);
  const envAppIdMatch = getFirstMatch(diskEnv, APP_ID_ALIASES);
  const envAppSecretMatch = getFirstMatch(diskEnv, APP_SECRET_ALIASES);
  const envVerifyTokenMatch = getFirstMatch(diskEnv, VERIFY_TOKEN_ALIASES);

  // 2. Database detection
  let dbAcc: any = null;
  if (isDatabaseConnected) {
    try {
      const accounts = await db.select().from(whatsappAccounts).limit(1);
      dbAcc = accounts[0] || null;
    } catch (err: any) {
      // Quiet notice
    }
  }

  // 3. Local store detection
  const localAcc = localStore.account;

  // Resolve Access Token
  let accessToken = '';
  let tokenSource: 'env' | 'database' | 'localStore' | 'none' = 'none';

  if (envTokenMatch.value) {
    accessToken = envTokenMatch.value;
    tokenSource = 'env';
  } else if (dbAcc?.accessToken) {
    accessToken = dbAcc.accessToken;
    tokenSource = 'database';
  } else if (localAcc?.accessToken) {
    accessToken = localAcc.accessToken;
    tokenSource = 'localStore';
  }

  // Resolve Phone Number ID
  let phoneNumberId = '';
  let phoneIdSource: 'env' | 'database' | 'localStore' | 'none' = 'none';

  if (envPhoneIdMatch.value) {
    phoneNumberId = envPhoneIdMatch.value;
    phoneIdSource = 'env';
  } else if (dbAcc?.phoneNumberId) {
    phoneNumberId = dbAcc.phoneNumberId;
    phoneIdSource = 'database';
  } else if (localAcc?.phoneNumberId) {
    phoneNumberId = localAcc.phoneNumberId;
    phoneIdSource = 'localStore';
  }

  // Resolve Business Account ID
  const businessAccountId =
    envWabaIdMatch.value || dbAcc?.businessAccountId || localAcc?.businessAccountId || '';

  // Resolve App ID & Secret
  const metaAppId = envAppIdMatch.value || dbAcc?.metaAppId || localAcc?.metaAppId || '';
  const metaAppSecret = envAppSecretMatch.value || dbAcc?.metaAppSecret || localAcc?.metaAppSecret || '';
  const webhookVerifyToken =
    envVerifyTokenMatch.value || dbAcc?.webhookVerifyToken || localAcc?.webhookVerifyToken || 'wa_verify_secret';
  const apiVersion = dbAcc?.apiVersion || localAcc?.apiVersion || 'v21.0';

  const hasToken = Boolean(accessToken && accessToken.length > 5);
  const hasPhoneId = Boolean(phoneNumberId && phoneNumberId.length > 3);
  const isFullyConfigured = hasToken && hasPhoneId;

  let errorNotes: string | undefined = undefined;
  if (!hasToken && !hasPhoneId) {
    errorNotes = 'WhatsApp credentials not configured. Provide Phone Number ID and Access Token.';
  } else if (hasToken && !hasPhoneId) {
    errorNotes =
      'Access token detected from environment (.env), but Phone Number ID is missing. Add META_PHONE_NUMBER_ID in .env or enter it in WhatsApp Settings.';
  } else if (!hasToken && hasPhoneId) {
    errorNotes = 'Phone Number ID is configured, but Access Token is missing. Add META_ACCESS_TOKEN in .env or in WhatsApp Settings.';
  }

  // Keep localStore in sync with resolved credentials so worker can access them instantly
  if (isFullyConfigured || hasToken || hasPhoneId) {
    if (!localStore.account) {
      localStore.account = {
        id: 1,
        metaAppId,
        metaAppSecret,
        businessAccountId,
        phoneNumberId,
        accessToken,
        webhookVerifyToken,
        apiVersion,
        displayPhoneNumber: dbAcc?.displayPhoneNumber || null,
        verifiedName: dbAcc?.verifiedName || (isFullyConfigured ? 'WhatsApp Business Account' : null),
        qualityRating: dbAcc?.qualityRating || (isFullyConfigured ? 'GREEN' : null),
        status: isFullyConfigured ? 'CONNECTED' : 'DISCONNECTED',
        lastTestedAt: dbAcc?.lastTestedAt || null,
        createdAt: new Date(),
        updatedAt: new Date(),
      };
    } else {
      if (accessToken && (!localStore.account.accessToken || tokenSource === 'env')) {
        localStore.account.accessToken = accessToken;
      }
      if (phoneNumberId && (!localStore.account.phoneNumberId || phoneIdSource === 'env')) {
        localStore.account.phoneNumberId = phoneNumberId;
      }
      if (businessAccountId && !localStore.account.businessAccountId) {
        localStore.account.businessAccountId = businessAccountId;
      }
      if (isFullyConfigured) {
        localStore.account.status = 'CONNECTED';
      }
    }
  }

  return {
    phoneNumberId,
    accessToken,
    businessAccountId,
    metaAppId,
    metaAppSecret,
    webhookVerifyToken,
    apiVersion,
    status: isFullyConfigured ? 'CONNECTED' : 'DISCONNECTED',
    hasToken,
    hasPhoneId,
    isFullyConfigured,
    tokenSource,
    phoneIdSource,
    errorNotes,
  };
}

  /**
   * Calculates the exact next birthday Date, target year, turning age, and day distance
   */
  static calculateNextBirthday(
    dobString: string,
    sendTimeStr: string = '09:00',
    now: Date = new Date()
  ): {
    birthYear: number;
    birthMonth: number;
    birthDay: number;
    targetYear: number;
    targetDate: Date;
    ageTurning: number;
    daysUntilBirthday: number;
    isToday: boolean;
    isThisMonth: boolean;
    formattedNextDate: string;
  } {
    const cleanDob = (dobString || '').trim();
    const parts = cleanDob.split(/[-/]/);
    const birthYear = parseInt(parts[0], 10) || 1990;
    const birthMonth = parseInt(parts[1], 10) || 1;
    const birthDay = parseInt(parts[2], 10) || 1;

    const timeParts = (sendTimeStr || '09:00').split(':');
    const sendHour = parseInt(timeParts[0], 10) || 9;
    const sendMinute = parseInt(timeParts[1], 10) || 0;

    const currentYear = now.getFullYear();

    // Check if birthday for current year is today or upcoming
    // Handle leap year Feb 29 on non-leap years (use Feb 28)
    const getSafeDate = (year: number, month: number, day: number, hour: number, min: number) => {
      if (month === 2 && day === 29) {
        const isLeap = (year % 4 === 0 && year % 100 !== 0) || year % 400 === 0;
        return new Date(year, 1, isLeap ? 29 : 28, hour, min, 0, 0);
      }
      return new Date(year, month - 1, day, hour, min, 0, 0);
    };

    let targetYear = currentYear;
    let targetDate = getSafeDate(currentYear, birthMonth, birthDay, sendHour, sendMinute);

    // If today is their birthday date (ignoring time)
    const isToday = now.getMonth() + 1 === birthMonth && now.getDate() === birthDay;
    const isThisMonth = now.getMonth() + 1 === birthMonth;

    // If birthday date has already completely passed this calendar year (and is not today)
    const todayZero = new Date(now.getFullYear(), now.getMonth(), now.getDate()).getTime();
    const thisYearBdayZero = new Date(currentYear, birthMonth - 1, birthDay).getTime();

    if (thisYearBdayZero < todayZero) {
      // Birthday already occurred earlier this year, target next year
      targetYear = currentYear + 1;
      targetDate = getSafeDate(targetYear, birthMonth, birthDay, sendHour, sendMinute);
    }

    const ageTurning = targetYear - birthYear;

    // Calculate days until birthday
    const nextBdayZero = new Date(targetDate.getFullYear(), targetDate.getMonth(), targetDate.getDate()).getTime();
    const diffMs = nextBdayZero - todayZero;
    const daysUntilBirthday = Math.max(0, Math.round(diffMs / (1000 * 60 * 60 * 24)));

    const yyyy = targetDate.getFullYear();
    const mm = String(targetDate.getMonth() + 1).padStart(2, '0');
    const dd = String(targetDate.getDate()).padStart(2, '0');
    const formattedNextDate = `${yyyy}-${mm}-${dd}`;

    return {
      birthYear,
      birthMonth,
      birthDay,
      targetYear,
      targetDate,
      ageTurning,
      daysUntilBirthday,
      isToday,
      isThisMonth,
      formattedNextDate,
    };
  }

  /**
   * Syncs the scheduled birthday automation for a specific contact.
   * Prevents duplicate messages for the same contact in the same year.
   */
  static async syncContactBirthdayAutomation(contactId: number): Promise<{
    scheduled: boolean;
    automationId?: number;
    message?: string;
  }> {
    const settings = this.getSettings();

    // 1. Fetch contact
    let contact: any = null;
    if (isDatabaseConnected) {
      try {
        const rows = await db.select().from(contacts).where(eq(contacts.id, contactId)).limit(1);
        contact = rows[0];
      } catch (err) {
        // Fallback to local store
      }
    }
    if (!contact) {
      contact = localStore.contacts.find((c) => c.id === contactId);
    }

    if (!contact) {
      return { scheduled: false, message: 'Contact not found' };
    }

    if (!contact.dob) {
      return { scheduled: false, message: 'Contact does not have a Date of Birth specified' };
    }

    // 2. Check if birthday automation is enabled for this contact and globally
    const isContactEnabled = contact.birthdayAutomationEnabled !== false;
    if (!settings.enabled || !isContactEnabled) {
      return { scheduled: false, message: 'Birthday automation is disabled' };
    }

    // 3. Compute target birthday
    const bday = this.calculateNextBirthday(contact.dob, settings.defaultSendTime);
    const targetYear = bday.targetYear;
    const targetDate = bday.targetDate;

    // 4. Check for existing automation for this contact in the target year to PREVENT DUPLICATES
    const existing = await this.findExistingBirthdayAutomation(contact.id, targetYear);
    if (existing) {
      // If already scheduled, ensure scheduled time is aligned
      if (existing.status === 'SCHEDULED' || existing.status === 'FAILED') {
        return {
          scheduled: true,
          automationId: existing.id,
          message: `Birthday wish already scheduled for ${targetYear}`,
        };
      }
      if (existing.status === 'SENT' || existing.status === 'DELIVERED' || existing.status === 'SIMULATED') {
        return {
          scheduled: false,
          automationId: existing.id,
          message: `Birthday wish already completed for ${targetYear}`,
        };
      }
    }

    // 5. Build template variables & message
    const personalizedMessage = settings.customMessage
      .replace(/{{name}}/g, contact.name)
      .replace(/{{age}}/g, String(bday.ageTurning));

    const templateVariables: Record<string, string> = {
      '1': contact.name,
      '2': String(bday.ageTurning),
      _isBirthday: 'true',
      _birthdayYear: String(targetYear),
      _contactName: contact.name,
      _customMessage: personalizedMessage,
    };

    // 6. Schedule automation using the existing scheduler table/store
    const effectiveTemplateId = settings.defaultTemplateId || 1;

    if (isDatabaseConnected) {
      try {
        const inserted = await db
          .insert(automations)
          .values({
            contactId: contact.id,
            templateId: effectiveTemplateId,
            variables: templateVariables,
            scheduledAt: targetDate,
            timezone: settings.timezone,
            status: 'SCHEDULED',
          })
          .returning();

        const newId = inserted[0]?.id;
        console.log(`[Wishflow] Scheduled birthday wish for ${contact.name} on ${targetDate.toISOString()} (Year: ${targetYear}, ID: ${newId})`);

class AutomationWorker {
  private isProcessing = false;
  private intervalTimer: NodeJS.Timeout | null = null;
  private pollIntervalMs = 5000; // Check every 5 seconds
  private lastWarningLogged = 0;

  start() {
    if (this.intervalTimer) return;
    console.log('[Worker] Starting scheduled WhatsApp automation worker...');

    // Run initial check after short delay
    setTimeout(() => {
      this.tick();
    }, 2000);

    this.intervalTimer = setInterval(() => {
      this.tick();
    }, this.pollIntervalMs);
  }

  stop() {
    if (this.intervalTimer) {
      clearInterval(this.intervalTimer);
      this.intervalTimer = null;
    }
    console.log('[Worker] Stopped WhatsApp automation worker');
  }

  async triggerNow(options?: { forceId?: number; forceAll?: boolean; simulate?: boolean }): Promise<WorkerRunResult> {
    return await this.processDueAutomations(options);
  }

  private async tick() {
    if (this.isProcessing) return;

    try {
      this.isProcessing = true;
      await this.processDueAutomations();
    } catch (err: any) {
      const msg = err?.message || String(err);
      if (
        msg.includes('SASL') ||
        msg.includes('password') ||
        msg.includes('client password') ||
        msg.includes('ECONNREFUSED') ||
        err?.code === '28P01'
      ) {
        const now = Date.now();
        if (now - this.lastWarningLogged > 60000) {
          console.warn('[Worker] PostgreSQL offline. Using local resilient store for automations.');
          this.lastWarningLogged = now;
        }
        return;
      }
      console.error('[Worker] Error in automation worker tick:', msg);
    } finally {
      this.isProcessing = false;
    }
  }

  private async processDueAutomations(options?: { forceId?: number; forceAll?: boolean; simulate?: boolean }): Promise<WorkerRunResult> {
    const now = new Date();
    const forceId = options?.forceId;
    const forceAll = options?.forceAll ?? false;
    const allowSimulation = options?.simulate ?? localStore.simulationMode;

    if (isDatabaseConnected) {
      try {
        return await this.processDatabaseAutomations(now, forceId, forceAll, allowSimulation);
      } catch (err: any) {
        console.warn('[Worker] Database query failed, checking local store:', err?.message);
      }
    }

    // Process from local memory store
    return await this.processLocalStoreAutomations(now, forceId, forceAll, allowSimulation);
  }

  private async processDatabaseAutomations(
    now: Date,
    forceId?: number,
    forceAll?: boolean,
    allowSimulation?: boolean
  ): Promise<WorkerRunResult> {
    // 1. Get active account credentials
    const accounts = await db.select().from(whatsappAccounts).limit(1);
    const account = accounts[0];

    const fiveMinutesAgo = new Date(Date.now() - 5 * 60 * 1000);

    let whereClause;
    if (forceId) {
      whereClause = eq(automations.id, forceId);
    } else if (forceAll) {
      whereClause = and(
        eq(automations.status, 'SCHEDULED'),
        or(isNull(automations.lockedAt), lte(automations.lockedAt, fiveMinutesAgo))
      );
    } else {
      whereClause = and(
        eq(automations.status, 'SCHEDULED'),
        lte(automations.scheduledAt, now),
        or(isNull(automations.lockedAt), lte(automations.lockedAt, fiveMinutesAgo))
      );
    }

    const dueJobs = await db
      .select({
        automation: automations,
        contact: contacts,
        template: messageTemplates,
      })
      .from(automations)
      .innerJoin(contacts, eq(automations.contactId, contacts.id))
      .innerJoin(messageTemplates, eq(automations.templateId, messageTemplates.id))
      .where(whereClause)
      .limit(10);

    if (dueJobs.length === 0) {
      return { processed: 0, successCount: 0, failureCount: 0, message: 'No due automations in database queue.' };
    }

    let successCount = 0;
    let failureCount = 0;
    const details: Array<{ id: number; contactName: string; status: string; error?: string }> = [];

    for (const job of dueJobs) {
      const { automation, contact, template } = job;

      // Lock job
      const lockResult = await db
        .update(automations)
        .set({
          status: 'PROCESSING',
          lockedAt: new Date(),
          updatedAt: new Date(),
        })
        .where(
          and(
            eq(automations.id, automation.id),
            eq(automations.status, 'SCHEDULED')
          )
        )
        .returning();

      if (lockResult.length === 0 && !forceId) {
        continue;
      }

      const res = await this.executeJob(automation, contact, template, account, allowSimulation);
      if (res.status === 'SENT') {
        successCount++;
      } else {
        failureCount++;
      }
      details.push({
        id: automation.id,
        contactName: contact.name,
        status: res.status,
        error: res.errorMessage,
      });
    }

    return {
      processed: dueJobs.length,
      successCount,
      failureCount,
      details,
      message: `Processed ${dueJobs.length} automation(s): ${successCount} sent, ${failureCount} failed.`,
    };
  }

import { Request, Response, NextFunction } from 'express';
import { adminAuth } from '../lib/firebase-admin.ts';
import { DecodedIdToken } from 'firebase-admin/auth';

export interface AuthRequest extends Request {
  user?: DecodedIdToken;
}

export const requireAuth = async (
  req: AuthRequest,
  res: Response,
  next: NextFunction
) => {
  const authHeader = req.headers.authorization;
  if (!authHeader || !authHeader.startsWith('Bearer ')) {
    return res.status(401).json({ error: 'Unauthorized: Missing token' });
  }

  const token = authHeader.split('Bearer ')[1];

  // Check for internal demo/admin session token
  if (token.startsWith('admin_session_token_')) {
    req.user = {
      uid: 'admin_navin_default',
      email: 'admin@whatsapp.business',
      name: 'System Administrator',
      auth_time: Math.floor(Date.now() / 1000),
      iss: 'https://securetoken.google.com/internal-admin',
      aud: 'internal-admin',
      sub: 'admin_navin_default',
      iat: Math.floor(Date.now() / 1000),
      exp: Math.floor(Date.now() / 1000) + 86400 * 30,
      firebase: { identities: {}, sign_in_provider: 'custom' },
    } as DecodedIdToken;
    return next();
  }

  try {
    const decodedToken = await adminAuth.verifyIdToken(token);
    req.user = decodedToken;
    next();
  } catch (error) {
    console.error('Error verifying Firebase ID token:', error);
    return res.status(401).json({ error: 'Unauthorized: Invalid or expired token' });
  }
};
